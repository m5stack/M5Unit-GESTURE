79inline float idle_time_to_hz(const uint16_t idle_time)

90inline uint16_t hz_to_idle_time(const float hz)
91{
92 if (hz <= 0.0f) {
93 return 0;
94 }
95 float v = 31250.0f / hz - 77.0f;
96 if (v < 0.0f) {
97 return 0;
98 }
99 if (v > 65535.0f) {
100 return 65535;
101 }
102 return static_cast<uint16_t>(v + 0.5f);
103}
